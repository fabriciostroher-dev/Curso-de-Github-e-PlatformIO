\begin{frame}[fragile]{Arquivos de configuração do Wokwi}
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \textbf{\texttt{wokwi.toml}}
            \begin{itemize}
                \item Diz ao Wokwi onde está o firmware compilado
                \item Fica na raiz do projeto, ao lado do \texttt{platformio.ini}
            \end{itemize}
            \begin{verbatim}
[wokwi]
version = 1
firmware = '.pio/build/uno/firmware.hex'
elf = '.pio/build/uno/firmware.elf'
            \end{verbatim}
        \end{column}
        \begin{column}{0.5\textwidth}
            \textbf{\texttt{diagram.json}}
            \begin{itemize}
                \item Descreve os componentes e as conexões do circuito
                \item Pode ser criado no editor online do Wokwi ou manualmente
                \item Validação: \texttt{wokwi-cli lint}
            \end{itemize}
            \begin{verbatim}
{
  "version": 1,
  "editor": "wokwi",
  "parts": [ ... ],
  "connections": [ ... ]
}
            \end{verbatim}
        \end{column}
    \end{columns}
    \vfill
    \footnotesize{No projeto do curso: \texttt{wokwi.toml} aponta para \texttt{.pio/build/uno/firmware.hex}; \texttt{diagram.json} contém o Arduino Uno, 2 DS18B20, LCD 20x4 I2C, 2 botões e resistor de 4,7 k$\Omega$.}
\end{frame}

    \end{block}

        \item O Wokwi recarrega automaticamente o novo firmware a cada \textit{build}
        \item Sensores DS18B20 podem ser ajustados na simulação (temperatura)
        \item Botões virtuais permitem testar a navegação entre telas
        \item Monitor serial do Wokwi auxilia na depuração
    \end{itemize}

\begin{frame}{Boas práticas de refatoração observadas}
    \begin{itemize}
        \item Separar responsabilidades: hardware, sensores, interface e navegação
        \item Evitar \texttt{delay()} longo; usar \texttt{millis()} para temporização
        \item Usar conversão não bloqueante: \texttt{setWaitForConversion(false)}
        \item Encapsular endereços e leituras em classes
        \item Reutilizar funções de desenho e impressão de endereço
        \item Versionar cada etapa com mensagens de commit claras
        \item Manter o \texttt{loop()} curto e legível
    \end{itemize}
\end{frame}

\section{GitHub}

\begin{frame}{GitHub: conceitos essenciais}
    \begin{itemize}
        \item \textbf{Repositório}: projeto versionado
        \item \textbf{Commit}: registro de alterações
        \item \textbf{Branch}: linha de desenvolvimento
        \item \textbf{Remote}: repositório remoto no GitHub
        \item \textbf{Push/Pull}: envio e recebimento de alterações
        \item \textbf{Issue}: registro de tarefa ou problema
        \item \textbf{Pull Request}: proposta de integração de mudanças
    \end{itemize}
    \vfill
    \begin{block}{No curso}
        Cada versão do projeto de temperatura pode ser um commit ou uma branch, documentando a evolução do código.
    \end{block}
\end{frame}

\begin{frame}[fragile]{Integração PlatformIO + GitHub}
    \begin{itemize}
        \item Versionar código-fonte, \texttt{platformio.ini}, bibliotecas locais e documentação
        \item Ignorar artefatos de compilação e arquivos locais:
    \end{itemize}
    \begin{block}{\texttt{.gitignore} sugerido}
        \begin{verbatim}
.pio/
.vscode/
*.log
        \end{verbatim}
    \end{block}
    \begin{itemize}
        \item Criar \texttt{README.md} com:
        \begin{itemize}
            \item descrição do projeto
            \item hardware utilizado
            \item bibliotecas e como compilar
            \item imagens ou diagrama de ligação
            \item versões/etapas didáticas
        \end{itemize}
        \item Usar commits pequenos e descritivos
    \end{itemize}
\end{frame}
